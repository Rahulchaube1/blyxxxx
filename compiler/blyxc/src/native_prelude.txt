// Runtime operations linked into a generated Blyx executable.
type Vector = std::rc::Rc<[f64]>;
trait Number { fn numeric(self) -> f64; }
impl Number for i64 { fn numeric(self) -> f64 { self as f64 } }
impl Number for f64 { fn numeric(self) -> f64 { self } }
fn number(n: impl Number) -> f64 { n.numeric() }
fn boolean(b: bool) -> bool { b }
fn len(v: Vector) -> i64 { v.len() as i64 }
fn sum(v: Vector) -> f64 { v.iter().sum() }
fn dot(a: Vector, b: Vector) -> f64 {
    if a.len() != b.len() { fail("dot requires vectors of equal length") }
    a.iter().zip(b.iter()).map(|(x,y)| x*y).sum()
}
fn at(v: Vector, i: i64) -> f64 {
    usize::try_from(i).ok().and_then(|i| v.get(i)).copied()
        .unwrap_or_else(|| fail("vector index out of bounds"))
}
fn fail(msg: &str) -> ! { eprintln!("runtime error: {msg}"); std::process::exit(4) }
trait Printable { fn emit(&self); }
macro_rules! printable {
    ($($t:ty),*) => { $(impl Printable for $t { fn emit(&self) { print!("{}", self) } })* }
}
printable!(i64, f64, bool, String);
impl Printable for () { fn emit(&self) {} }
impl Printable for Vector { fn emit(&self) { print!("{:?}", self) } }
fn emit(value: impl Printable) { value.emit() }
